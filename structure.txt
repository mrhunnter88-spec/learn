1. PASSIVE SUBDOMAIN ENUMERATION
# subfinder (fastest passive)
subfinder -d hackerone.com -all -o subs1.txt

# amass (deep passive)
amass enum -passive -d hackerone.com -o subs2.txt

# assetfinder (lightweight)
assetfinder --subs-only hackerone.com > subs3.txt

# findomain (certificate transparency)
findomain -t hackerone.com -q

# sublist3r (search engines)
sublist3r -d hackerone.com -o subs4.txt

# theHarvester (OSINT)
cd ~/theHarvester && PYTHONPATH=. python3 -m theHarvester -d hackerone.com -b bing,google -f subs5.txt
2. ACTIVE DNS ENUMERATION
# dnsrecon (bruteforce + zone transfer)

massdns -r resolvers.txt -t A subs.txt -o S

# puredns (wildcard filtering)
puredns resolve subs.txt -r resolvers.txt
puredns brute-force /usr/share/wordlists/dnsmap.txt hackerone.com
